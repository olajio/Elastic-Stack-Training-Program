        def getCheckedFields(def source, def rules_param) {
          def all_field_names = new HashSet();
          for (entry in rules_param.entrySet()) {
            all_field_names.add(entry.getKey());
          }
          return all_field_names;
        }
        ArrayList getFieldRules(def source, def rules_param) {
          ArrayList fields = new ArrayList();
          for (entry in rules_param.entrySet()) {
            def field_name = entry.getKey();
            def field = source;
            for (String seg : field_name.splitOnToken(".")) {
              if (field instanceof Map && !field.containsKey(seg)) { field = null; break; }
              field = field[seg];
            }
            fields.add(['field_name': field_name, 'field_value': field, 'cfg': entry.getValue()])
          }
          return fields;
        }
        boolean fieldRuleHandleRequired(def field_rule, def required_error) {
          if(!field_rule["cfg"].containsKey("required")) return true;

          def field_value = field_rule["field_value"];
          def field_name = field_rule["field_name"];

          boolean is_valid = !fieldCheckMissing(field_value);
          if (!is_valid) { 
            required_error.add(["field": field_name]); 
          }
          return is_valid;
        }
        boolean fieldRuleHandleExpectedType(def field_rule, def datatype_error, boolean legacy_mode) {
          if(!field_rule["cfg"].containsKey("expected_type")) return true;

          def field_value = field_rule["field_value"];
          def field_name = field_rule["field_name"];
          def expected_type = field_rule["cfg"]["expected_type"];

          def message = null;
          def supplied_type = getSimpleType(field_value);
          // The old functions seem to expect float for double.
          if (legacy_mode && supplied_type == "double") supplied_type = "float";

          if (expected_type == "date") {
            if (!validDateFormat(field_value)) {
              message = "Invalid Date Format. An ISO-8601 Date/Time value is expected.";
            }
          }else if (expected_type != supplied_type){
            message = "Invalid Data Type. " + expected_type + " value is expected. But " + supplied_type + " type is supplied.";
          }

          boolean is_valid = (message == null);
          if (!is_valid) {
            datatype_error.add(["field": field_name, "value": field_value, "message": message]);
          }
          return is_valid;
        }
        boolean fieldRuleHandleEnumerated(def field_rule, def enumerated_error) {
          if(!field_rule["cfg"].containsKey("enumerated")) return true;

          def field_value = field_rule["field_value"];
          def field_name = field_rule["field_name"];
          def expected_values = field_rule["cfg"]["enumerated"];
          def message = "Invalid value. Please reference the published data target documentation list of accepted values for this field.";

          boolean is_valid = expected_values.contains(normalizeString(field_value));
          if (!is_valid) {
            enumerated_error.add(["field": field_name, "value": field_value, "message": message ]); 
          }
          return is_valid;
        }
